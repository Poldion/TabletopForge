# Firmware-Konfiguration des Spieltisch-Kiosks von Tabletop Forge (Pi Zero 2 W, 64 Bit)
start_file=start.elf
fixup_file=fixup.dat
kernel=Image
arm_64bit=1

# Systemplatz: Kernel, Gerätebaum, Overlays und cmdline.txt liegen in a/ bzw. b/ (Updates über das Netz).
# Der Kiosk-Agent schaltet hier um; ein neues System startet zuerst einmalig zur Probe über tryboot.txt.
os_prefix=a/

# Schneller, ruhiger Start
boot_delay=0
disable_splash=1
disable_overscan=1
camera_auto_detect=0
display_auto_detect=0

# Grafik per KMS; Grafikspeicher (CMA) 256 MB – nötig für das GPU-Zeichnen im Browser.
# Ungenutzter CMA-Speicher steht dem System weiter zur Verfügung.
gpu_mem=32
dtoverlay=vc4-kms-v3d,cma-256
max_framebuffers=1

# Grafikchip schneller (Standard 300/250 MHz): gemessen ~13 % mehr Bilder/s beim Verschieben der Karte,
# ohne Unterspannung, 2–3 °C wärmer. Der Prozessortakt bleibt unverändert.
v3d_freq=400
core_freq=400
core_freq_min=400

# Nicht gebraucht: Bluetooth, Ton
dtoverlay=disable-bt
dtparam=audio=off
